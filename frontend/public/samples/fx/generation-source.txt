async function download(url,file){const r=await fetch(url);if(!r.ok)throw Error(r.status);const b=Buffer.from(await r.arrayBuffer());if(b.length<100||b.toString('utf8',0,100).includes('<html'))throw Error('Invalid audio');fs.writeFileSync(root+'/'+file,b);}
function synth(p){const rate=44100,n=Math.ceil(p.duration*rate),b=Buffer.alloc(44+n*2);b.write('RIFF');b.writeUInt32LE(36+n*2,4);b.write('WAVEfmt ',8);b.writeUInt32LE(16,16);b.writeUInt16LE(1,20);b.writeUInt16LE(1,22);b.writeUInt32LE(rate,24);b.writeUInt32LE(rate*2,28);b.writeUInt16LE(2,32);b.writeUInt16LE(16,34);b.write('data',36);b.writeUInt32LE(n*2,40);let phase=0,seed=1234567,held=0;for(let i=0;i<n;i++){const t=i/rate,u=i/n;seed=(Math.imul(seed,1664525)+1013904223)>>>0;const noise=seed/2147483648-1;const freq=p.start*Math.pow(p.end/p.start,u);phase+=2*Math.PI*freq/rate;let v=Math.sin(phase+(p.fm||0)*Math.sin(phase*.37));if(p.noise)v=(1-p.noise)*v+p.noise*noise;if(p.crush){if(i%p.crush===0)held=Math.round(v*7)/7;v=held;}let env=p.reverse?Math.pow(u,2):Math.exp(-5*u);env*=Math.min(1,t/.003,(p.duration-t)/.008);if(p.gate)env*=Math.sin(2*Math.PI*p.gate*t)>-.2?1:.03;b.writeInt16LE(Math.round(Math.max(-1,Math.min(1,v*env*.75))*32767),44+i*2);}return b;}
